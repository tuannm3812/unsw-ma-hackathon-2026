# log. requirements.txt intentionally uses lower bounds (>=) so a normal
# `pip install -r requirements.txt` stays installable as new package

# cases and warnings on the same code (this project hit exactly that
# category of bug more than once during development - see the
# collaboration log's warning-cleanup and reproducibility rounds).
#
# Not meant to replace requirements.txt for normal installs. Use this only
# when you need to reproduce a committed artifact bit-for-bit:
#
#   pip install -r requirements-lock.txt
#

# individual packages themselves enforce; Python 3.9+ per README).
#

# against requirements.txt, not a substring grep - avoids false positives
# like "notebook_shim" matching "notebook"):
#
#   python3 -c "
#   import subprocess
#   names = {l.split('>=')[0].strip().lower() for l in open('requirements.txt') if l.strip()}
#   for line in subprocess.run(['pip', 'freeze'], capture_output=True, text=True).stdout.splitlines():
#       if '==' in line and line.split('==')[0].lower() in names:

#   "

# notebook re-execution, PDF re-render) before committing the new lock.
